from flask import Flask, request, send_file, jsonify
import zipfile
import json
import os
import tempfile
import traceback
from lottie.exporters import export_tgs
import logging

app = Flask(__name__)

# Logging setup
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

@app.route('/lottie/', methods=['POST'])
def convert_lottie():
    try:
        if 'zipfile' not in request.files:
            return jsonify({"error": "No zipfile provided"}), 400
        
        zip_file = request.files['zipfile']
        if not zip_file.filename.endswith('.zip'):
            return jsonify({"error": "Only ZIP files allowed"}), 400
        
        # Create temp directory
        with tempfile.TemporaryDirectory() as work_dir:
            input_zip = os.path.join(work_dir, 'input.zip')
            output_zip = os.path.join(work_dir, 'output.zip')
            
            # Save uploaded zip
            zip_file.save(input_zip)
            
            # Extract input zip
            with zipfile.ZipFile(input_zip, 'r') as zin:
                zin.extractall(work_dir)
            
            # Find and convert JSON files to TGS
            tgs_files = []
            json_count = 0
            
            for root, _, files in os.walk(work_dir):
                for file in files:
                    if file.endswith('.json'):
                        json_count += 1
                        json_path = os.path.join(root, file)
                        tgs_path = json_path.replace('.json', '.tgs')
                        
                        try:
                            # Load Lottie JSON
                            with open(json_path, 'r', encoding='utf-8') as f:
                                lottie_data = json.load(f)
                            
                            # Export to Telegram TGS format
                            export_tgs(lottie_data, tgs_path)
                            tgs_files.append(tgs_path)
                            logger.info(f"Converted: {file} -> {os.path.basename(tgs_path)}")
                            
                        except Exception as e:
                            logger.error(f"Failed to convert {file}: {str(e)}")
                            continue
            
            if not tgs_files:
                return jsonify({"error": f"No valid TGS files created (processed {json_count} JSONs)"}), 400
            
            # Create output ZIP with TGS files
            with zipfile.ZipFile(output_zip, 'w', zipfile.ZIP_DEFLATED) as zout:
                for tgs in tgs_files:
                    zout.write(tgs, os.path.basename(tgs))
            
            logger.info(f"Conversion complete: {len(tgs_files)} TGS files created")
            return send_file(output_zip, mimetype='application/zip', as_attachment=True)
    
    except Exception as e:
        logger.error(f"API Error: {str(e)}")
        return jsonify({"error": "Internal server error"}), 500

@app.route('/health', methods=['GET'])
def health():
    return jsonify({"status": "healthy", "service": "lottie-converter"}), 200

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5001, debug=False)
